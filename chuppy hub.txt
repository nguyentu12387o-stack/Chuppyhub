-- 🔧 Helpers
local function GetCharacter()
    local char = LP.Character
    if char and char:FindFirstChild("HumanoidRootPart") then return char end
    return nil
end
local function GetHRP()
    local char = GetCharacter()
    return char and char:FindFirstChild("HumanoidRootPart")
end
local function GetHumanoid()
    local char = GetCharacter()
    return char and char:FindFirstChildOfClass("Humanoid")
end
local function IsAlive()
    local hum = GetHumanoid()
    return hum and hum.Health > 0
end
local function GetLevel()
    local data = LP:FindFirstChild("Data")
    return data and data:FindFirstChild("Level") and data.Level.Value or 0
end
local function GetBeli()
    local data = LP:FindFirstChild("Data")
    return data and data:FindFirstChild("Beli") and data.Beli.Value or 0
end
local function GetFragments()
    local data = LP:FindFirstChild("Data")
    return data and data:FindFirstChild("Fragments") and data.Fragments.Value or 0
end
local function GetCommF()
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    return remotes and remotes:FindFirstChild("CommF_")
end
local function InvokeServer(...)
    local commF = GetCommF()
    if not commF then return nil end
    local args = { ... }
    local unpackFn = table.unpack or unpack
    local ok, res = pcall(function()
        return commF:InvokeServer(unpackFn(args))
    end)
    if not ok then
        warn("[MasterHub] InvokeServer error:", res)
    end
    return ok and res or nil
end

local function CreateScreenGui()
    local sg = Instance.new("ScreenGui")
    sg.Name = "MasterHub_" .. math.random(1000, 9999)
    sg.ResetOnSpawn = false
    sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    pcall(function() sg.Parent = CoreGui end)
    if not sg.Parent then sg.Parent = LP:WaitForChild("PlayerGui") end
    return sg
end

            function section:AddSlider(cfg)
                local name  = cfg.Name or "Slider"
                local min   = cfg.Min or 0
                local max   = cfg.Max or 100
                local value = cfg.Default or min
                local cb    = cfg.Callback or function() end

                local frame = createElement(50)
                local lbl = Instance.new("TextLabel")
                lbl.Size = UDim2.new(1, -60, 0, 18)
                lbl.Position = UDim2.new(0, 10, 0, 3)
                lbl.BackgroundTransparency = 1
                lbl.Text = name .. ": " .. value
                lbl.TextColor3 = Color3.fromRGB(200, 200, 200)
                lbl.TextSize = 12
                lbl.Font = Enum.Font.Gotham
                lbl.TextXAlignment = Enum.TextXAlignment.Left
                lbl.Parent = frame

                local bg = Instance.new("Frame")
                bg.Size = UDim2.new(1, -20, 0, 10)
                bg.Position = UDim2.new(0, 10, 0, 28)
                bg.BackgroundColor3 = Color3.fromRGB(60, 60, 80)
                bg.Parent = frame
                Instance.new("UICorner", bg).CornerRadius = UDim.new(0, 5)

                local fill = Instance.new("Frame")
                fill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
                fill.BackgroundColor3 = Color3.fromRGB(80, 150, 255)
                fill.Parent = bg
                Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 5)

                local dragging = false
                bg.InputBegan:Connect(function(i)
                    if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true end
                end)
                UserInputService.InputEnded:Connect(function(i)
                    if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
                end)
                UserInputService.InputChanged:Connect(function(i)
                    if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then
                        local rel = math.clamp((i.Position.X - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
                        value = math.floor(min + (max - min) * rel)
                        fill.Size = UDim2.new(rel, 0, 1, 0)
                        lbl.Text = name .. ": " .. value
                        pcall(cb, value)
                    end
                end)
            end

            return section
        end

        table.insert(self.tabs, tab)
        if #self.tabs == 1 then
            tabBtn.BackgroundColor3 = Color3.fromRGB(80, 120, 200)
            tabContent.Visible = true
            self.currentTab = tab
        end
        return tab
    end

    return self
end

local function CheckQuest()
    local myLevel = GetLevel()
    for _, data in ipairs(QuestDatabase) do
        if myLevel >= data.minLevel and myLevel <= data.maxLevel then
            if data.world == CurrentWorld then
                QuestData.Mon          = data.mon
                QuestData.NameQuest    = data.quest
                QuestData.LevelQuest   = data.level
                QuestData.NameMon      = data.mon
                QuestData.CFrameQuest  = data.questCF
                QuestData.CFrameMon    = data.monCF
                return data
            end
        end
    end
    return nil
end

local function TweenTo(targetCF, speed)
    speed = speed or 300
    local hrp = GetHRP()
    if not hrp then return end
    if TweenManager.currentTween then
        TweenManager.currentTween:Cancel()
        TweenManager.currentTween = nil
    end
    local dist = (hrp.Position - targetCF.Position).Magnitude
    if dist < 3 then return end
    local time = dist / speed
    local tw = TweenService:Create(hrp,
        TweenInfo.new(time, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut),
        { CFrame = targetCF })
    TweenManager.currentTween = tw
    tw:Play()
    tw.Completed:Connect(function(s)
        if s == Enum.PlaybackState.Completed then TweenManager.currentTween = nil end
    end)
    return tw
end

local function StopTween()
    if TweenManager.currentTween then
        TweenManager.currentTween:Cancel()
        TweenManager.currentTween = nil
    end
end

local function EquipWeapon(weaponName)
    if not weaponName then return end
    local char = GetCharacter()
    if not char then return end
    local hum = GetHumanoid()
    if not hum then return end

    local tool = LP.Backpack:FindFirstChild(weaponName)
    if tool then
        pcall(function() hum:EquipTool(tool) end)
        return true
    end

    local current = char:FindFirstChildOfClass("Tool")
    if current and current.Name ~= weaponName then
        pcall(function() current.Parent = LP.Backpack end)
        task.wait(0.1)
        tool = LP.Backpack:FindFirstChild(weaponName)
        if tool then
            hum:EquipTool(tool)
            return true
        end
    end
    return false
end

local function FindMonster(monsterName)
    -- Ưu tiên Enemies folder
    local enemies = workspace:FindFirstChild("Enemies")
    if enemies then
        for _, v in pairs(enemies:GetChildren()) do
            if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid")
               and v.Humanoid.Health > 0 then
                if not monsterName
                   or string.find(string.lower(v.Name), string.lower(monsterName)) then
                    return v
                end
            end
        end
    end

    -- Fallback: tìm trong Workspace gốc (mob thường trú Sea 1)
    if monsterName then
        for _, v in pairs(workspace:GetChildren()) do
            if v:IsA("Model")
               and string.find(string.lower(v.Name), string.lower(monsterName))
               and v:FindFirstChild("HumanoidRootPart")
               and v:FindFirstChild("Humanoid")
               and v.Humanoid.Health > 0 then
                return v
            end
        end
    end

    -- Fallback cuối: ReplicatedStorage (mob chưa spawn)
    if monsterName then
        local rsMob = ReplicatedStorage:FindFirstChild(monsterName)
        if rsMob and rsMob:FindFirstChild("HumanoidRootPart") then
            return rsMob
        end
    end

    return nil
end

local function FindNearestMonster(maxDistance)
    maxDistance = maxDistance or 5000
    local hrp = GetHRP()
    if not hrp then return nil end
    local enemies = workspace:FindFirstChild("Enemies")
    if not enemies then return nil end
    local nearest, shortest = nil, maxDistance
    for _, v in pairs(enemies:GetChildren()) do
        if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid")
           and v.Humanoid.Health > 0 then
            local d = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
            if d < shortest then
                shortest = d
                nearest = v
            end
        end
    end
    return nearest
end

local NoclipConnection = nil

            local char = GetCharacter()
            if char then
                for _, p in pairs(char:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
            end
        end)
    else
        if NoclipConnection then
            NoclipConnection:Disconnect()
            NoclipConnection = nil
        end
    end
end

local FlyConnection = nil
local function EnableFly(enable, speed)
    speed = speed or 200
    local hrp = GetHRP()
    if not hrp then return end
    if enable then
        local bv = hrp:FindFirstChild("FlyVelocity") or Instance.new("BodyVelocity")
        bv.Name = "FlyVelocity"
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bv.Velocity = Vector3.zero
        bv.Parent = hrp

        if FlyConnection then FlyConnection:Disconnect() end
        FlyConnection = RunService.RenderStepped:Connect(function()
            local bv2 = hrp:FindFirstChild("FlyVelocity")
            if not bv2 then return end
            local cam = workspace.CurrentCamera
            local dir = Vector3.zero
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
            bv2.Velocity = dir.Magnitude > 0 and dir.Unit * speed or Vector3.zero
        end)
    else
        if FlyConnection then FlyConnection:Disconnect(); FlyConnection = nil end
        if hrp:FindFirstChild("FlyVelocity") then hrp.FlyVelocity:Destroy() end
    end
end

local function ServerHop()
    local ok, jobId = pcall(function()
        local url = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local res = HttpService:JSONDecode(game:HttpGet(url))
        if res and res.data then
            for _, srv in ipairs(res.data) do
                if srv.playing < srv.maxPlayers and srv.id ~= game.JobId then
                    return srv.id
                end
            end
        end
        return nil
    end)
    if ok and jobId then
        TeleportService:TeleportToPlaceInstance(PlaceId, jobId, LP)
    else
        pcall(function()
            StarterGui:SetCore("SendNotification", {
                Title = "Lỗi", Text = "Không tìm thấy server", Duration = 3,
            })
        end)
    end
end

-- 📋 Lấy trạng thái quest từ UI
local function GetQuestStatus()
    local pg = LP:FindFirstChild("PlayerGui")
    if not pg then return { hasQuest = false } end
    local main = pg:FindFirstChild("Main")
    if not main then return { hasQuest = false } end
    local questUI = main:FindFirstChild("Quest")
    if not questUI then return { hasQuest = false } end

    local title = ""
    if questUI.Visible then
        local container = questUI:FindFirstChild("Container")
        if container then
            local qTitle = container:FindFirstChild("QuestTitle")
            if qTitle then
                local titleLbl = qTitle:FindFirstChild("Title")
                if titleLbl then title = titleLbl.Text or "" end
            end
        end
    end
    return { hasQuest = questUI.Visible, title = title, ui = questUI }
end

    local t0 = tick()
    while model and model.Parent and hum.Health > 0 and tick() - t0 < 120 do
        DoAttack()
        task.wait(0.1)
    end
    if tween then tween:Cancel() end
end

    for _, plr in ipairs(Players:GetPlayers()) do

                end

-- Hook __namecall để bắn tự dính (chỉ khi executor hỗ trợ)
pcall(function()
    if not (getrawmetatable and setreadonly and newcclosure and getnamecallmethod) then return end
    local mt    = getrawmetatable(game)
    local oldNC = mt.__namecall
    if not oldNC then return end
    setreadonly(mt, false)
    mt.__namecall = newcclosure(function(self, ...)
        local args   = { ... }
        local method = getnamecallmethod()
        if PVP.Aimbot and PVP.Target and method == "FireServer"
           and typeof(self) == "Instance" and self:IsA("RemoteEvent") then
            local tChar = PVP.Target.Character
            local hrp   = tChar and tChar:FindFirstChild("HumanoidRootPart")
            if hrp then
                for i, v in ipairs(args) do
                    if typeof(v) == "Vector3" then args[i] = hrp.Position end
                    if typeof(v) == "CFrame"  then args[i] = hrp.CFrame   end
                end
            end
        end
        return oldNC(self, unpack(args))
    end)
    setreadonly(mt, true)
end)
