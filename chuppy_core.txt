​local function CheckCooldown(key, duration)

return true
end
return false
end

​local Monster = {}
​function Monster.Find(name)
local search = name and string.lower(name) or nil
local enemiesFolder = Workspace:FindFirstChild("Enemies")
​if enemiesFolder then
for _, v in ipairs(enemiesFolder:GetChildren()) do
if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
if not search or string.find(string.lower(v.Name), search, 1, true) then
return v
end
end
end
end
​if search then
for _, v in ipairs(Workspace:GetChildren()) do
if v:IsA("Model") and string.find(string.lower(v.Name), search, 1, true)
and v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
return v
end
end
end
​if name then
local rsMob = ReplicatedStorage:FindFirstChild(name)
if rsMob and rsMob:FindFirstChild("HumanoidRootPart") then
return rsMob
end
end
return nil
end
​function Monster.FindNearest(maxDistance)
maxDistance = maxDistance or 5000
local hrp = PlayerUtils.GetHRP()
if not hrp then return nil end
​local enemiesFolder = Workspace:FindFirstChild("Enemies")
if not enemiesFolder then return nil end
​local nearest, shortest = nil, maxDistance
for _, v in ipairs(enemiesFolder:GetChildren()) do
if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
local dist = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
if dist < shortest then
shortest = dist
nearest = v
end
end
end
return nearest
end
​function Monster.FindInRange(range)
range = range or 300
local hrp = PlayerUtils.GetHRP()
if not hrp then return {} end
​local list = {}
local enemiesFolder = Workspace:FindFirstChild("Enemies")
if not enemiesFolder then return list end
​for _, v in ipairs(enemiesFolder:GetChildren()) do
if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
local dist = (v.HumanoidRootPart.Position - hrp.Position).Magnitude
if dist <= range then
table.insert(list, { mob = v, distance = dist })
end
end
end
table.sort(list, function(a, b) return a.distance < b.distance end)
return list
end
​-- Optimized Bring Logic with BodyVelocity Anchor & Collision Override
function Monster.Bring(mob, targetCF)
if not mob or not mob:FindFirstChild("HumanoidRootPart") then return end
pcall(function()
local hrp = mob.HumanoidRootPart
local hum = mob:FindFirstChildOfClass("Humanoid")
​hrp.CFrame = targetCF
hrp.CanCollide = false
if hum then
hum.WalkSpeed = 0
hum.JumpPower = 0
end
​local bv = hrp:FindFirstChild("BringLock")
if not bv then
bv = Instance.new("BodyVelocity")
bv.Name = "BringLock"
bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
bv.Velocity = Vector3.zero
bv.Parent = hrp
end
end)
end
​function Monster.BringAll(mobs, targetCF, radius)
radius = radius or State.BringMobDistance
local hrp = PlayerUtils.GetHRP()
if not hrp or not targetCF then return end
​for _, entry in ipairs(mobs or {}) do
local m = entry.mob or entry
if m and m:FindFirstChild("HumanoidRootPart") then
if (m.HumanoidRootPart.Position - hrp.Position).Magnitude <= radius then
Monster.Bring(m, targetCF)
end
end
end
end
​function Monster.IsAlive(mob)
if not mob or not mob.Parent then return false end
local hum = mob:FindFirstChildOfClass("Humanoid")
return hum and hum.Health > 0
end

webhookSending = true
​local item = table.remove(webhookQueue, 1)

​local KeybindManager = {}

​function KeybindManager.Bind(action, keyCode, callback)
if not keyCode or not callback then return end
binds[keyCode] = binds[keyCode] or {}
table.insert(binds[keyCode], { action = action, callback = callback })
end
​function KeybindManager.Unbind(action)
for keyCode, list in pairs(binds) do
for i = #list, 1, -1 do
if list[i].action == action then
table.remove(list, i)
end
end
end
end

end
end
end))
